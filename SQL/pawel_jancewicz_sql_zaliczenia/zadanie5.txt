A) SELECT MAX(Payment) AS MaxIncome,
       MIN(Payment) AS MinIncome,
       ROUND(AVG(Payment), 2) AS AvgIncome,
       SUM(Payment) AS AllProfessorsPaymentsSummed,
       COUNT(ProfessorId) AS ProfessorsCount
  FROM Professors;

       P.ProgramName,
       SUM(S.Tuition * 6) AS ThreeYearsTuitionsSumPerProgram

       JOIN
       Programs P ON S.ProgramId = P.ProgramId

D) SELECT D.DepartmentName,
       AVG(P.Payment) AS AveragePayment
  FROM Departments D
       LEFT JOIN
       Professors P ON D.DepartmentId = P.DepartmentId
 GROUP BY D.DepartmentName
HAVING AveragePayment > 7000;
