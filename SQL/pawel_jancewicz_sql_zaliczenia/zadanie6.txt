A) WITH StudentsAge AS (
    SELECT StudentId,
           ProgramId,
           CAST (STRFTIME('%Y.%m%d', 'now') - STRFTIME('%Y.%m%d', DateOfBirth) AS INT) AS StudentAge
      FROM Students
),
AverageStudentAgeOnCSDepartment AS (
    SELECT D.DepartmentId,
           D.DepartmentName,
           ROUND(AVG(SA.StudentAge), 2) AS AvgStudentAge
      FROM StudentsAge SA
           JOIN
           Programs P ON SA.ProgramId = P.ProgramId
           JOIN
           Departments D ON D.DepartmentId = P.DepartmentId
     GROUP BY D.DepartmentName
    HAVING D.DepartmentId = 1
)
SELECT S.FirstName,
       S.LastName,
       S.DateOfBirth
  FROM Students S
       JOIN
       StudentsAge SA ON S.StudentId = SA.StudentId
 WHERE SA.StudentAge < (
                           SELECT AvgStudentAge
                             FROM AverageStudentAgeOnCSDepartment
                       );

B) WITH AvgDepartmentPayment AS (
    SELECT D.DepartmentId,
           D.DepartmentName,
           AVG(P.Payment) AS AverageDepartmentPayment
      FROM Departments D
           LEFT JOIN
           Professors P ON D.DepartmentId = P.DepartmentId
     GROUP BY D.DepartmentName
)
SELECT P.FirstName,
       P.LastName,
       P.Payment,
       ADP.DepartmentName
  FROM Professors P
       LEFT JOIN
       AvgDepartmentPayment ADP ON P.DepartmentId = ADP.DepartmentId
 WHERE P.Payment < ADP.AverageDepartmentPayment
 ORDER BY P.LastName ASC,
          P.FirstName ASC;

C) WITH DepartmentsStudentsData AS (
    SELECT D.DepartmentId,
           D.DepartmentName,
           COUNT( * ) AS StudentsCountInDepartment,
           SUM(S.Tuition * 2) AS AnnualDepartmentIncomeFromTuitions
      FROM Departments D
           JOIN
           Programs P ON D.DepartmentId = P.DepartmentId
           JOIN
           Students S ON P.ProgramId = S.ProgramId
     GROUP BY D.DepartmentId
),
DepartmentsProfessorsData AS (
    SELECT D.DepartmentId,
           D.DepartmentName,
           COUNT( * ) AS ProfessorsCountInDepartment,
           SUM(P.Payment * 12) AS AnnualPaymentsForProfessorsInDepartment
      FROM Professors P
           JOIN
           Departments D ON P.DepartmentId = D.DepartmentId
     GROUP BY D.DepartmentId
)
SELECT DSD.DepartmentName,
       DSD.StudentsCountInDepartment,
       DPD.ProfessorsCountInDepartment,
       DSD.AnnualDepartmentIncomeFromTuitions,
       DPD.AnnualPaymentsForProfessorsInDepartment,
       DSD.AnnualDepartmentIncomeFromTuitions - DPD.AnnualPaymentsForProfessorsInDepartment AS AnnualIncome
  FROM DepartmentsStudentsData DSD
       JOIN
       DepartmentsProfessorsData DPD ON DSD.DepartmentId = DPD.DepartmentId
 WHERE AnnualIncome > 0 AND
       DPD.ProfessorsCountInDepartment > 1;
