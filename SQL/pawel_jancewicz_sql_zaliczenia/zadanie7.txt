B) CREATE VIEW EarlyGenZ AS
WITH StudentsData AS (
        SELECT S.FirstName,
               S.LastName,
               C.CourseName,
               CAST (STRFTIME('%Y', DateOfBirth) AS INT) AS YearOfBirth
          FROM Students S
               JOIN
               Enrollments E ON S.StudentId = E.StudentId
               JOIN
               Courses C ON C.CourseId = E.CourseId
    )
    SELECT FirstName,
           LastName,
           CourseName
      FROM StudentsData
     WHERE YearOfBirth BETWEEN 1997 AND 2000;

C) CREATE VIEW SocialDepartments AS
    SELECT P.FirstName,
           P.LastName,
           P.Payment * 1.15 AS Payment
      FROM Professors P
           JOIN
           Departments D ON P.DepartmentId = D.DepartmentId
     WHERE D.DepartmentName = 'Psychology' OR
           D.DepartmentName = 'Sociology';
